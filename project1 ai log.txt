四、 核心 JS：javascript.js
这是驱动天体运行、悬停发光、地月系视角切换以及导航触发过渡动画的核心逻辑：
window.addEventListener('DOMContentLoaded', () => {
   const canvas = document.getElementById('solarCanvas');
   if (!canvas) return; // 非 Home 页面则退出

   const ctx = canvas.getContext('2d');
   const backBtn = document.getElementById('backToSolarBtn');

   let width = (canvas.width = window.innerWidth);
   let height = (canvas.height = window.innerHeight);

   window.addEventListener('resize', () => {
       width = canvas.width = window.innerWidth;
       height = canvas.height = window.innerHeight;
   });

   // 状态管理：'SOLAR'（太阳系视角） | 'EARTH_MOON'（地月系视角）
   let currentMode = 'SOLAR'; 
   let hoveredEntity = null;
   let transitionState = { active: false, targetUrl: '', progress: 0 };

   // 太阳系天体定义
   const sun = { x: 0, y: 0, radius: 45, color: '#f97316' };
   const asteroidBelt = { minRadius: 180, maxRadius: 210, particles: [] };

   // 初始化小行星带粒子
   for (let i = 0; i < 250; i++) {
       const angle = Math.random() * Math.PI * 2;
       const dist = asteroidBelt.minRadius + Math.random() * (asteroidBelt.maxRadius - asteroidBelt.minRadius);
       asteroidBelt.particles.push({ angle, dist, speed: 0.0005 + Math.random() * 0.001, size: Math.random() * 1.8 + 0.5 });
   }

   // 8大行星数据
   const planets = [
       { name: 'Mercury', distance: 80, radius: 4, color: '#a1a1aa', speed: 0.015, angle: 0, interactive: false },
       { name: 'Venus', distance: 120, radius: 7, color: '#fbbf24', speed: 0.01, angle: 1, interactive: false },
       { name: 'Earth', distance: 160, radius: 9, color: '#38bdf8', speed: 0.008, angle: 2, interactive: true },
       { name: 'Mars', distance: 240, radius: 6, color: '#f87171', speed: 0.006, angle: 3, interactive: false },
       { name: 'Jupiter', distance: 290, radius: 18, color: '#eab308', speed: 0.004, angle: 4, interactive: false },
       { name: 'Saturn', distance: 350, radius: 14, color: '#fde047', speed: 0.003, angle: 5, interactive: false },
       { name: 'Uranus', distance: 400, radius: 11, color: '#22d3ee', speed: 0.002, angle: 6, interactive: false },
       { name: 'Neptune', distance: 450, radius: 10, color: '#60a5fa', speed: 0.001, angle: 7, interactive: false }
   ];

   // 地月系视角数据
   const earthMoonSystem = {
       earth: { x: 0, y: 0, radius: 40, color: '#38bdf8' },
       moon: { distance: 120, radius: 12, color: '#e2e8f0', angle: 0, speed: 0.02 }
   };

   // 主绘制循环
   function animate() {
       ctx.clearRect(0, 0, width, height);
       const centerX = width / 2;
       const centerY = height / 2;

       if (currentMode === 'SOLAR') {
           drawSolarSystem(centerX, centerY);
       } else if (currentMode === 'EARTH_MOON') {
           drawEarthMoonSystem(centerX, centerY);
       }

       // 绘制过渡亮起动画
       if (transitionState.active) {
           drawTransitionEffect(centerX, centerY);
       }

       requestAnimationFrame(animate);
   }

   // 1. 绘制太阳系
   function drawSolarSystem(cx, cy) {
       // 绘制太阳
       ctx.beginPath();
       ctx.arc(cx, cy, sun.radius, 0, Math.PI * 2);
       ctx.fillStyle = sun.color;
       ctx.shadowBlur = (hoveredEntity === 'Sun' || transitionState.target === 'Sun') ? 60 : 30;
       ctx.shadowColor = '#f97316';
       ctx.fill();
       ctx.shadowBlur = 0;

       // 绘制小行星带
       const isAsteroidHovered = hoveredEntity === 'AsteroidBelt' || transitionState.target === 'AsteroidBelt';
       asteroidBelt.particles.forEach(p => {
           p.angle += p.speed;
           const px = cx + Math.cos(p.angle) * p.dist;
           const py = cy + Math.sin(p.angle) * p.dist;
           ctx.beginPath();
           ctx.arc(px, py, p.size, 0, Math.PI * 2);
           ctx.fillStyle = isAsteroidHovered ? '#38bdf8' : '#64748b';
           ctx.fill();
       });

       // 绘制行星及其轨道
       planets.forEach(p => {
           p.angle += p.speed;
           const px = cx + Math.cos(p.angle) * p.distance;
           const py = cy + Math.sin(p.angle) * p.distance;
           p.x = px; p.y = py;

           // 轨道线
           ctx.beginPath();
           ctx.arc(cx, cy, p.distance, 0, Math.PI * 2);
           ctx.strokeStyle = 'rgba(255, 255, 255, 0.05)';
           ctx.stroke();

           // 行星本体
           ctx.beginPath();
           ctx.arc(px, py, p.radius, 0, Math.PI * 2);
           ctx.fillStyle = p.color;
           if (hoveredEntity === p.name) {
               ctx.shadowBlur = 20;
               ctx.shadowColor = p.color;
           }
           ctx.fill();
           ctx.shadowBlur = 0;
       });
   }

   // 2. 绘制地月系次级视图
   function drawEarthMoonSystem(cx, cy) {
       // 绘制地球 (Projects Part 1)
       earthMoonSystem.earth.x = cx;
       earthMoonSystem.earth.y = cy;
       ctx.beginPath();
       ctx.arc(cx, cy, earthMoonSystem.earth.radius, 0, Math.PI * 2);
       ctx.fillStyle = earthMoonSystem.earth.color;
       ctx.shadowBlur = hoveredEntity === 'Earth_Sub' ? 40 : 15;
       ctx.shadowColor = '#38bdf8';
       ctx.fill();

       // 绘制月球 (About)
       const em = earthMoonSystem.moon;
       em.angle += em.speed;
       const mx = cx + Math.cos(em.angle) * em.distance;
       const my = cy + Math.sin(em.angle) * em.distance;
       em.x = mx; em.y = my;

       ctx.beginPath();
       ctx.arc(mx, my, em.radius, 0, Math.PI * 2);
       ctx.fillStyle = em.color;
       ctx.shadowBlur = hoveredEntity === 'Moon_Sub' ? 30 : 10;
       ctx.shadowColor = '#e2e8f0';
       ctx.fill();
       ctx.shadowBlur = 0;
   }

   // 鼠标悬停检测
   canvas.addEventListener('mousemove', (e) => {
       const rect = canvas.getBoundingClientRect();
       const mx = e.clientX - rect.left;
       const my = e.clientY - rect.top;
       const cx = width / 2;
       const cy = height / 2;

       hoveredEntity = null;

       if (currentMode === 'SOLAR') {
           // 检查太阳
           if (Math.hypot(mx - cx, my - cy) < sun.radius) {
               hoveredEntity = 'Sun';
               return;
           }
           // 检查小行星带
           const distFromCenter = Math.hypot(mx - cx, my - cy);
           if (distFromCenter >= asteroidBelt.minRadius && distFromCenter <= asteroidBelt.maxRadius) {
               hoveredEntity = 'AsteroidBelt';
               return;
           }
           // 检查行星
           planets.forEach(p => {
               if (Math.hypot(mx - p.x, my - p.y) < p.radius + 5) {
                   hoveredEntity = p.name;
               }
           });
       } else if (currentMode === 'EARTH_MOON') {
           if (Math.hypot(mx - cx, my - cy) < earthMoonSystem.earth.radius) {
               hoveredEntity = 'Earth_Sub';
           } else if (Math.hypot(mx - earthMoonSystem.moon.x, my - earthMoonSystem.moon.y) < earthMoonSystem.moon.radius) {
               hoveredEntity = 'Moon_Sub';
           }
       }
   });

   // 鼠标点击事件绑定
   canvas.addEventListener('click', () => {
       if (currentMode === 'SOLAR') {
           if (hoveredEntity === 'Sun') triggerTransition('contact.html', 'Sun');
           else if (hoveredEntity === 'Earth') {

   function drawSolarSystem(cx, cy) {
       // 太阳日冕发光
       const glowRadius = sun.radius + Math.sin(sunPulse) * 5 + (hoveredEntity === 'Sun' ? 20 : 0);
       const sunGlow = ctx.createRadialGradient(cx, cy, sun.radius * 0.4, cx, cy, glowRadius + 30);
       sunGlow.addColorStop(0, '#fdba74');
       sunGlow.addColorStop(0.5, 'rgba(249, 115, 22, 0.45)');
       sunGlow.addColorStop(1, 'rgba(249, 115, 22, 0)');
       
       ctx.beginPath();
       ctx.arc(cx, cy, glowRadius + 30, 0, Math.PI * 2);
       ctx.fillStyle = sunGlow;
       ctx.fill();

       ctx.beginPath();
       ctx.arc(cx, cy, sun.radius, 0, Math.PI * 2);
       ctx.fillStyle = sun.color;
       ctx.fill();

       // 小行星带
       const isAsteroidHovered = hoveredEntity === 'AsteroidBelt';
       asteroidBelt.particles.forEach(p => {
           p.angle += p.speed;
           const px = cx + Math.cos(p.angle) * p.dist;
           const py = cy + Math.sin(p.angle) * p.dist;
           ctx.beginPath();
           ctx.arc(px, py, p.size, 0, Math.PI * 2);
           ctx.fillStyle = isAsteroidHovered ? '#38bdf8' : p.color;
           ctx.fill();
       });

       // 8大行星
       planets.forEach(p => {
           p.angle += p.speed;
           const px = cx + Math.cos(p.angle) * p.distance;
           const py = cy + Math.sin(p.angle) * p.distance;
           p.x = px; p.y = py;

               ctx.fillStyle = 'rgba(255, 255, 255, 0.2)';
               ctx.fill();

               ctx.translate(px, py);
               ctx.rotate(0.4);

               ctx.strokeStyle = 'rgba(254, 240, 138, 0.7)';
               ctx.lineWidth = 3;
               ctx.stroke();
               ctx.restore();
           }

           ctx.beginPath();

           ctx.fillStyle = p.color;
           ctx.fill();

           if (p.hasStripes) {
               ctx.save();

               ctx.arc(px, py, p.radius, 0, Math.PI * 2);
               ctx.clip();

这是基于 Three.js 重构的 3D 渲染引擎代码，包含了真实的立体光源、自转公转动画、光照阴影、宇宙星云背景以及 3D 鼠标拾取（Raycaster）：
window.addEventListener('DOMContentLoaded', () => {
   const container = document.getElementById('canvas-container');
   const backBtn = document.getElementById('backToSolarBtn');
   if (!container) return;

   // 1. 场景、相机与渲染器
   const scene = new THREE.Scene();
   const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
   camera.position.set(0, 15, 35);

   const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
   renderer.setSize(window.innerWidth, window.innerHeight);
   renderer.setPixelRatio(window.devicePixelRatio);
   renderer.shadowMap.enabled = true;
   container.appendChild(renderer.domElement);

   // 2. 纹理加载器 (内置公网高清贴图)
   const textureLoader = new THREE.TextureLoader();
   const textures = {
       sun: textureLoader.load('https://raw.githubusercontent.com/mrdoob/three.js/dev/examples/textures/planets/sun.jpg'),
       earth: textureLoader.load('https://raw.githubusercontent.com/mrdoob/three.js/dev/examples/textures/planets/earth_atmo_2048.jpg'),
       moon: textureLoader.load('https://raw.githubusercontent.com/mrdoob/three.js/dev/examples/textures/planets/moon_1024.jpg')
   };

   // 3. 星空背景粒体
   const starsGeo = new THREE.BufferGeometry();
   const starsCount = 2000;
   const starPositions = new Float32Array(starsCount * 3);
   for(let i=0; i<starsCount*3; i++) {
       starPositions[i] = (Math.random() - 0.5) * 300;
   }
   starsGeo.setAttribute('position', new THREE.BufferAttribute(starPositions, 3));
   const starsMat = new THREE.PointsMaterial({ color: 0xffffff, size: 0.7 });
   const starField = new THREE.Points(starsGeo, starsMat);
   scene.add(starField);

   // 4. 光源定义
   const ambientLight = new THREE.AmbientLight(0x333333); // 宇宙暗环境光
   scene.add(ambientLight);

   const sunLight = new THREE.PointLight(0xffffff, 2, 300); // 太阳发出的点光源
   scene.add(sunLight);

   // 5. 3D 天体构建
   // 太阳 (发光体)
   const sunGeo = new THREE.SphereGeometry(5, 32, 32);
   const sunMat = new THREE.MeshBasicMaterial({ map: textures.sun });
   const sunMesh = new THREE.Mesh(sunGeo, sunMat);
   scene.add(sunMesh);

   // 地球
   const earthGeo = new THREE.SphereGeometry(2, 32, 32);
   const earthMat = new THREE.MeshStandardMaterial({ map: textures.earth, roughness: 0.6 });
   const earthMesh = new THREE.Mesh(earthGeo, earthMat);
   earthMesh.position.set(18, 0, 0);
   scene.add(earthMesh);

   // 月球
   const moonGeo = new THREE.SphereGeometry(0.6, 16, 16);
   const moonMat = new THREE.MeshStandardMaterial({ map: textures.moon, roughness: 0.8 });
   const moonMesh = new THREE.Mesh(moonGeo, moonMat);
   earthMesh.add(moonMesh); // 将月球绑定为地球的子物体，实现围绕地球转

   // 6. 交互射线与状态
   const raycaster = new THREE.Raycaster();
   const mouse = new THREE.Vector2();
   let hoveredObject = null;
   let mode = 'SOLAR'; // 'SOLAR' | 'EARTH_MOON'

   // 7. 动画主循环
   let angleEarth = 0;
   let angleMoon = 0;

   function animate() {
       requestAnimationFrame(animate);

       // 自转
       sunMesh.rotation.y += 0.002;
       earthMesh.rotation.y += 0.01;
       moonMesh.rotation.y += 0.005;

       // 地球绕太阳公转
       if (mode === 'SOLAR') {
           angleEarth += 0.005;
           earthMesh.position.x = Math.cos(angleEarth) * 20;
           earthMesh.position.z = Math.sin(angleEarth) * 20;
       }

       // 月球绕地球公转
       angleMoon += 0.02;
       moonMesh.position.x = Math.cos(angleMoon) * 5;
       moonMesh.position.z = Math.sin(angleMoon) * 5;

       // 星空极慢旋转
       starField.rotation.y += 0.0002;

       renderer.render(scene, camera);
   }

   // 8. 鼠标悬停检测 (3D 射线拾取)
   window.addEventListener('mousemove', (e) => {
       mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
       mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;

       raycaster.setFromCamera(mouse, camera);
       const intersects = raycaster.intersectObjects([sunMesh, earthMesh, moonMesh]);

       if (intersects.length > 0) {
           document.body.style.cursor = 'pointer';
           hoveredObject = intersects[0].object;
       } else {
           document.body.style.cursor = 'default';
           hoveredObject = null;
       }
   });

   // 9. 点击交互与相机镜头切换
   window.addEventListener('click', () => {
       if (!hoveredObject) return;

       if (hoveredObject === sunMesh) {
           window.location.href = 'contact.html';
       } else if (hoveredObject === earthMesh && mode === 'SOLAR') {
           // 视角拉近进入地月系
           mode = 'EARTH_MOON';
           backBtn.classList.remove('hidden');
           
           // 相机平滑聚焦到地球（简单平移）
           camera.position.set(earthMesh.position.x, 3, earthMesh.position.z + 12);
           camera.lookAt(earthMesh.position);
       } else if (hoveredObject === earthMesh && mode === 'EARTH_MOON') {
           window.location.href = 'projects.html';
       } else if (hoveredObject === moonMesh && mode === 'EARTH_MOON') {
           window.location.href = 'about.html';
       }
   });

   // 返回太阳系视角
   backBtn.addEventListener('click', () => {
       mode = 'SOLAR';
       camera.position.set(0, 15, 35);
       camera.lookAt(0, 0, 0);
       backBtn.classList.add('hidden');
   });

   // 窗口尺寸自适应
   window.addEventListener('resize', () => {
       camera.aspect = window.innerWidth / window.innerHeight;
       camera.updateProjectionMatrix();
       renderer.setSize(window.innerWidth, window.innerHeight);
   });

   animate();
});

   const canvas = document.getElementById('solarCanvas');
   const backBtn = document.getElementById('backToSolarBtn');
   if (!canvas) return;

   const ctx = canvas.getContext('2d');

   // 1. 适配屏幕分辨率
   function resizeCanvas() {
       canvas.width = window.innerWidth;
       canvas.height = window.innerHeight;
   }
   resizeCanvas();
   window.addEventListener('resize', resizeCanvas);

   // 2. 全局状态与滚轮缩放/拖拽平移变量
   let mode = 'SOLAR'; // 'SOLAR' | 'EARTH_MOON'
   let hoveredObject = null;

   let offsetY = 0;
   let isDragging = false;
   let startX = 0, startY = 0;

   // 3. 天体数据调整
   const sun = { x: 0, y: 0, radius: 50, color: '#fbbf24', glow: '#f59e0b', name: 'Sun', func: 'Click to Contact' };

       dist: 260,       // 拉大月球公转轨道半径
       angle: 0, 
       speed: 0.008, 
       color: '#9ca3af', 
       name: 'Moon', 
       func: 'Click to About Me'

       { name: 'Mercury', radius: 7, dist: 90, angle: 1, speed: 0.006, color: '#94a3b8' },
       { name: 'Venus', radius: 12, dist: 140, angle: 2, speed: 0.004, color: '#fde047' },
       earth,
       { name: 'Mars', radius: 10, dist: 300, angle: 4, speed: 0.002, color: '#ef4444' },
       { name: 'Jupiter', radius: 32, dist: 480, angle: 5, speed: 0.001, color: '#eab308' },
       { name: 'Saturn', radius: 26, dist: 600, angle: 0.5, speed: 0.0007, color: '#fef08a', hasRing: true }
   ];

   // 小行星带粒子数据
   const asteroidBelt = Array.from({ length: 300 }, () => ({
       dist: 340 + Math.random() * 80,
       angle: Math.random() * Math.PI * 2,
       speed: 0.001 + Math.random() * 0.0008,
       size: 1 + Math.random() * 2,
       alpha: 0.3 + Math.random() * 0.5
   }));

   // 背景星空粒子
   const stars = Array.from({ length: 200 }, () => ({
       x: Math.random() * window.innerWidth,
       y: Math.random() * window.innerHeight,
       size: Math.random() * 1.5,
       alpha: Math.random()
   }));

   // 4. 渲染主循环
   function render() {
       ctx.clearRect(0, 0, canvas.width, canvas.height);
       const centerX = canvas.width / 2;
       const centerY = canvas.height / 2;

       // A. 绘制静态背景星空

           ctx.fillStyle = `rgba(255, 255, 255, ${star.alpha})`;
           ctx.beginPath();
           ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);

           ctx.save();

           sun.y = centerY + offsetY;

           ctx.arc(earth.x, earth.y, 120, 0, Math.PI * 2);
           ctx.fill();

           surfaceGrad.addColorStop(0, 'rgba(255, 255, 255, 0.35)');
           surfaceGrad.addColorStop(0.8, 'rgba(0, 0, 0, 0.45)');

           ctx.beginPath();
           ctx.arc(earth.x, earth.y, 120, 0, Math.PI * 2);
           ctx.fill();

           moon.x = centerX + Math.cos(moon.angle) * moon.dist;
           moon.y = centerY + Math.sin(moon.angle) * moon.dist;

           // 月球轨道线
           ctx.strokeStyle = 'rgba(255, 255, 255, 0.15)';
           ctx.lineWidth = 1.5;
           ctx.beginPath();
           ctx.arc(centerX, centerY, moon.dist, 0, Math.PI * 2);
           ctx.stroke();

           // 月球实体及纹理渐变
           const moonGrad = ctx.createRadialGradient(moon.x - 5, moon.y - 5, 2, moon.x, moon.y, moon.radius);
           moonGrad.addColorStop(0, '#f3f4f6');

           ctx.fill();
       }

       // 绘制悬停提示文字
       if (hoveredObject) {
           ctx.fillStyle = '#ffffff';
           ctx.font = 'bold 15px sans-serif';
           ctx.textAlign = 'center';
           const displayRadius = (mode === 'EARTH_MOON' && hoveredObject === earth) ? 120 : (hoveredObject.radius * (mode === 'SOLAR' ? zoomLevel : 1));
           ctx.fillText(hoveredObject.name, hoveredObject.x, hoveredObject.y - displayRadius - 15);
       }

       requestAnimationFrame(render);
   }

   // 5. 交互：滚轮缩放 (仅太阳系模式)
   canvas.addEventListener('wheel', (e) => {
       if (mode !== 'SOLAR') return;
       e.preventDefault();

       const zoomFactor = e.deltaY < 0 ? 1.1 : 0.9;
       zoomLevel = Math.min(Math.max(0.3, zoomLevel * zoomFactor), 3.0);
   }, { passive: false });

   // 6. 交互：鼠标拖拽画布平移
   canvas.addEventListener('mousedown', (e) => {
       if (mode !== 'SOLAR') return;
       isDragging = true;
       startX = e.clientX - offsetX;
       startY = e.clientY - offsetY;
   });

   canvas.addEventListener('mouseup', () => isDragging = false);
   canvas.addEventListener('mouseleave', () => isDragging = false);

   // 7. 交互：鼠标移动与拾取检测
   canvas.addEventListener('mousemove', (e) => {
       const rect = canvas.getBoundingClientRect();
       const mouseX = e.clientX - rect.left;
       const mouseY = e.clientY - rect.top;

       if (isDragging && mode === 'SOLAR') {
           offsetX = e.clientX - startX;
           offsetY = e.clientY - startY;
           return;
       }

       let targets = (mode === 'SOLAR') ? [sun, ...planets] : [earth, moon];
       hoveredObject = null;

       targets.forEach(obj => {
           const currentRadius = (mode === 'EARTH_MOON' && obj === earth) ? 120 : (obj.radius * (mode === 'SOLAR' ? zoomLevel : 1));
           const dist = Math.hypot(mouseX - obj.x, mouseY - obj.y);
           if (dist < Math.max(currentRadius, 15) + 10) {
               hoveredObject = obj;
           }
       });

       canvas.style.cursor = hoveredObject ? 'pointer' : (isDragging ? 'grabbing' : 'default');
   });

   // 8. 点击跳转
   canvas.addEventListener('click', () => {
       if (!hoveredObject) return;

       if (mode === 'SOLAR') {
           if (hoveredObject === sun) {
               window.location.href = 'contact.html';
           } else if (hoveredObject === earth) {
               mode = 'EARTH_MOON';
               backBtn.classList.remove('hidden');
           }
       } else if (mode === 'EARTH_MOON') {
           if (hoveredObject === earth) {
               window.location.href = 'projects.html';
           } else if (hoveredObject === moon) {
               window.location.href = 'about.html';
           }
       }
   });

   // 9. 返回太阳系大视角
   backBtn.addEventListener('click', () => {
       mode = 'SOLAR';
       backBtn.classList.add('hidden');
       zoomLevel = 1.0;
       offsetX = 0;
       offsetY = 0;
   });

   render();
});
