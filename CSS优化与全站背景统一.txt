   function drawSolarSystem(cx, cy) {
       // 太阳日冕发光
       const glowRadius = sun.radius + Math.sin(sunPulse) * 5 + (hoveredEntity === 'Sun' ? 20 : 0);
       const sunGlow = ctx.createRadialGradient(cx, cy, sun.radius * 0.4, cx, cy, glowRadius + 30);
       sunGlow.addColorStop(0, '#fdba74');
       sunGlow.addColorStop(0.5, 'rgba(249, 115, 22, 0.45)');
       sunGlow.addColorStop(1, 'rgba(249, 115, 22, 0)');
       
       ctx.beginPath();
       ctx.arc(cx, cy, glowRadius + 30, 0, Math.PI * 2);
       ctx.fillStyle = sunGlow;
       ctx.fill();

       ctx.beginPath();
       ctx.arc(cx, cy, sun.radius, 0, Math.PI * 2);
       ctx.fillStyle = sun.color;
       ctx.fill();

       // 小行星带
       const isAsteroidHovered = hoveredEntity === 'AsteroidBelt';
       asteroidBelt.particles.forEach(p => {
           p.angle += p.speed;
           const px = cx + Math.cos(p.angle) * p.dist;
           const py = cy + Math.sin(p.angle) * p.dist;
           ctx.beginPath();
           ctx.arc(px, py, p.size, 0, Math.PI * 2);
           ctx.fillStyle = isAsteroidHovered ? '#38bdf8' : p.color;
           ctx.fill();
       });

       // 8大行星
       planets.forEach(p => {
           p.angle += p.speed;
           const px = cx + Math.cos(p.angle) * p.distance;
           const py = cy + Math.sin(p.angle) * p.distance;
           p.x = px; p.y = py;

           ctx.beginPath();
           ctx.arc(cx, cy, p.distance, 0, Math.PI * 2);
           ctx.strokeStyle = 'rgba(255, 255, 255, 0.07)';
           ctx.stroke();

           if (hoveredEntity === p.name) {
               ctx.beginPath();
               ctx.arc(px, py, p.radius + 10, 0, Math.PI * 2);
               ctx.fillStyle = 'rgba(255, 255, 255, 0.2)';
               ctx.fill();
           }

           if (p.hasRings) {
               ctx.save();
               ctx.translate(px, py);
               ctx.rotate(0.4);
               ctx.beginPath();
               ctx.ellipse(0, 0, p.radius * 2.3, p.radius * 0.7, 0, 0, Math.PI * 2);
               ctx.strokeStyle = 'rgba(254, 240, 138, 0.7)';
               ctx.lineWidth = 3;
               ctx.stroke();
               ctx.restore();
           }

           ctx.beginPath();
           ctx.arc(px, py, p.radius, 0, Math.PI * 2);
           ctx.fillStyle = p.color;
           ctx.fill();

           if (p.hasStripes) {
               ctx.save();
               ctx.beginPath();
               ctx.arc(px, py, p.radius, 0, Math.PI * 2);
               ctx.clip();
               ctx.fillStyle = '#ca8a04';
               ctx.fillRect(px - p.radius, py - 4, p.radius * 2, 3);
               ctx.fillRect(px - p.radius, py + 3, p.radius * 2, 4);
               ctx.restore();
           }
       });
   }

二、 如何微调地标位置或增加地区？
1. 调整百分比坐标： 在 .map-marker 的 style="top: 38%; left: 79%;" 中，top 代表纬度位置（越往下值越大），left 代表经度位置（越往右值越大）。通过微调这两个百分比即可精确平移光标。
2. 更换地标： 你可以把光标内部的 Emoji（如 🗽, 🏯, 🕰️, 🗻）替换为任何你喜欢的 SVG 图标、图片或 Emoji。
3. 增加/删除地点： 直接复制一份 <div class="map-marker">...</div> 并在 JavaScript 的 projectData 中加上对应的文字介绍即可！

   <script>
       window.addEventListener('DOMContentLoaded', () => {
           const canvas = document.getElementById('asteroidCanvas');
           if (!canvas) return;

           const ctx = canvas.getContext('2d');
           let width = (canvas.width = window.innerWidth);
           let height = (canvas.height = window.innerHeight);

           window.addEventListener('resize', () => {
               width = canvas.width = window.innerWidth;
               height = canvas.height = window.innerHeight;
           });

           let mousePos = { x: -1000, y: -1000 };
           let hoveredAsteroid = null;

           // ==========================================
           // 1. 无交互小陨石群 (背景流)
           // ==========================================
           const smallAsteroids = [];
           const SMALL_COUNT = 120; // 小陨石数量

           for (let i = 0; i < SMALL_COUNT; i++) {
               smallAsteroids.push({
                   x: Math.random() * width,
                   y: Math.random() * height,
                   radius: Math.random() * 2.5 + 0.8,
                   speedY: Math.random() * 0.6 + 0.2, // 慢慢从上往下漂
                   speedX: (Math.random() - 0.5) * 0.15,
                   color: Math.random() > 0.4 ? '#94a3b8' : '#64748b',
                   opacity: Math.random() * 0.6 + 0.3
               });
           }

           // ==========================================
           // 2. 大陨石 (项目卡片数据，可随时添加/调节数量)
           // ==========================================
           const largeAsteroids = [
               {
                   id: 'portfolio',
                   title: 'Interactive 3D/2D Portfolio',
                   subtitle: 'Click to return Home',
                   x: width * 0.5,       // 初始居中偏上方
                   y: height * 0.35,
                   radius: 55,           // 大陨石尺寸
                   speedY: 0.35,         // 下落速度
                   targetUrl: 'index.html',
                   color: '#38bdf8',
                   rotation: 0,
                   rotationSpeed: 0.005,
                   // 不规则多边形节点（让陨石看起来像石头而非完美圆形）
                   shapeVertices: [
                       { angle: 0, scale: 1.1 },
                       { angle: Math.PI * 0.3, scale: 0.85 },
                       { angle: Math.PI * 0.6, scale: 1.05 },
                       { angle: Math.PI * 0.9, scale: 0.9 },
                       { angle: Math.PI * 1.2, scale: 1.15 },
                       { angle: Math.PI * 1.5, scale: 0.95 },
                       { angle: Math.PI * 1.8, scale: 1.02 }
                   ]
               }
               /* 后期可以在这里轻松扩展更多大陨石项目：
               ,{
                   id: 'project2',
                   title: 'Future Algorithm Engine',
                   subtitle: 'Coming soon...',
                   x: width * 0.25,
                   y: -100,
                   radius: 48,
                   speedY: 0.4,
                   targetUrl: '#',
                   color: '#fde047',
                   ...
               }
               */
           ];

           // ==========================================
           // 3. 渲染主循环
           // ==========================================
           function animate() {
               ctx.clearRect(0, 0, width, height);

               // --- 绘制无交互小陨石 ---
               smallAsteroids.forEach(p => {
                   p.y += p.speedY;
                   p.x += p.speedX;

                   // 飘出屏幕底部后重新回到顶部
                   if (p.y > height + 10) {
                       p.y = -10;
                       p.x = Math.random() * width;
                   }

                   ctx.beginPath();
                   ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
                   ctx.fillStyle = p.color;
                   ctx.globalAlpha = p.opacity;
                   ctx.fill();
                   ctx.globalAlpha = 1.0;
               });

               // --- 绘制大陨石 (交互项目) ---
               hoveredAsteroid = null;

               largeAsteroids.forEach(ast => {
                   // 碰撞/悬停检测
                   const dist = Math.hypot(mousePos.x - ast.x, mousePos.y - ast.y);
                   const isHovered = dist < ast.radius + 15;

                   if (isHovered) {
                       hoveredAsteroid = ast;
                   } else {
                       // 没悬停时正常下落和缓慢旋转
                       ast.y += ast.speedY;
                       ast.rotation += ast.rotationSpeed;

                       // 飘出屏幕后循环回到顶部
                       if (ast.y > height + ast.radius * 2) {
                           ast.y = -ast.radius * 2;
                           ast.x = Math.random() * (width - 200) + 100;
                       }
                   }

                   // 绘制大陨石主体
                   ctx.save();
                   ctx.translate(ast.x, ast.y);
                   ctx.rotate(ast.rotation);

                   // 悬停高亮光晕
                   if (isHovered) {
                       const glowGrad = ctx.createRadialGradient(0, 0, ast.radius * 0.8, 0, 0, ast.radius + 25);
                       glowGrad.addColorStop(0, 'rgba(56, 189, 248, 0.5)');
                       glowGrad.addColorStop(1, 'rgba(56, 189, 248, 0)');
                       ctx.beginPath();
                       ctx.arc(0, 0, ast.radius + 25, 0, Math.PI * 2);
                       ctx.fillStyle = glowGrad;
                       ctx.fill();
                   }

                   // 绘制不规则岩石形状
                   ctx.beginPath();
                   ast.shapeVertices.forEach((v, index) => {
                       const r = ast.radius * v.scale;
                       const vx = Math.cos(v.angle) * r;
                       const vy = Math.sin(v.angle) * r;
                       if (index === 0) ctx.moveTo(vx, vy);
                       else ctx.lineTo(vx, vy);
                   });
                   ctx.closePath();

                   // 岩石填充与边框
                   ctx.fillStyle = isHovered ? '#1e293b' : '#0f172a';
                   ctx.strokeStyle = isHovered ? '#38bdf8' : '#475569';
                   ctx.lineWidth = isHovered ? 3 : 2;
                   ctx.fill();
                   ctx.stroke();

                   // 陨石表面纹理凹坑
                   ctx.fillStyle = isHovered ? 'rgba(56, 189, 248, 0.2)' : 'rgba(255, 255, 255, 0.05)';
                   ctx.beginPath();
                   ctx.arc(-ast.radius * 0.3, -ast.radius * 0.2, ast.radius * 0.25, 0, Math.PI * 2);
                   ctx.arc(ast.radius * 0.2, ast.radius * 0.3, ast.radius * 0.2, 0, Math.PI * 2);
                   ctx.fill();

                   ctx.restore();

                   // 悬停时浮现项目名称 Title Label (在陨石右下方)
                   if (isHovered) {
                       drawAsteroidLabel(ast);
                   }
               });

               requestAnimationFrame(animate);
           }

           // 绘制悬停时的项目 Label 弹窗
           function drawAsteroidLabel(ast) {
               const lx = ast.x + ast.radius + 15;
               const ly = ast.y - 20;

               ctx.font = 'bold 16px "Inter", sans-serif';
               const titleWidth = ctx.measureText(ast.title).width;
               ctx.font = '13px "Inter", sans-serif';
               const subWidth = ctx.measureText(ast.subtitle).width;

               const boxWidth = Math.max(titleWidth, subWidth) + 30;
               const boxHeight = 56;

               // 悬停框背景
               ctx.fillStyle = 'rgba(15, 23, 42, 0.95)';
               ctx.strokeStyle = '#38bdf8';
               ctx.lineWidth = 1.5;
               ctx.beginPath();
               ctx.roundRect(lx, ly, boxWidth, boxHeight, 10);
               ctx.fill();
               ctx.stroke();

               // 标题与副标题文本
               ctx.fillStyle = '#ffffff';
               ctx.font = 'bold 15px "Inter", sans-serif';
               ctx.fillText(ast.title, lx + 15, ly + 24);

               ctx.fillStyle = '#38bdf8';
               ctx.font = '13px "Inter", sans-serif';
               ctx.fillText(ast.subtitle, lx + 15, ly + 44);
           }

           // ==========================================
           // 4. 事件监听 (鼠标移动与点击)
           // ==========================================
           canvas.addEventListener('mousemove', (e) => {
               const rect = canvas.getBoundingClientRect();
               mousePos.x = e.clientX - rect.left;
               mousePos.y = e.clientY - rect.top;

               // 悬停在有交互的大陨石上时改变鼠标手势
               if (hoveredAsteroid) {
                   canvas.style.cursor = 'pointer';
               } else {
                   canvas.style.cursor = 'default';
               }
           });

3. JavaScript 逻辑 (光标跟随 + 面板开关 + Dark Mode)

document.addEventListener('DOMContentLoaded', () => {
   // ==========================================
   // 1. 火箭光标跟随与喷火交互
   // ==========================================
   const rocket = document.getElementById('rocketCursor');

   if (rocket) {
       window.addEventListener('mousemove', (e) => {
           rocket.style.left = `${e.clientX}px`;
           rocket.style.top = `${e.clientY}px`;
       });

       // 监听可交互元素，触发火焰喷射
       const interactiveSelector = 'a, button, .map-marker, input, .switch, [role="button"]';
       
       document.addEventListener('mouseover', (e) => {
           if (e.target.closest(interactiveSelector)) {
               rocket.classList.add('hovering');
           }
       });

       document.addEventListener('mouseout', (e) => {
           if (e.target.closest(interactiveSelector)) {
               rocket.classList.remove('hovering');
           }
       });
   }

   // ==========================================
   // 2. 左上角 Logo 面板与 Dark Mode 切换
   // ==========================================
   const logoBtn = document.getElementById('logoControlBtn');
   const controlPanel = document.getElementById('controlPanel');
   const darkModeToggle = document.getElementById('darkModeToggle');

   // 展开 / 隐藏控制面板
   if (logoBtn && controlPanel) {
       logoBtn.addEventListener('click', (e) => {
           e.stopPropagation();
           const isVisible = controlPanel.style.display === 'block';
           controlPanel.style.display = isVisible ? 'none' : 'block';
       });

       // 点击页面其他区域自动收起面板
       document.addEventListener('click', (e) => {
           if (!controlPanel.contains(e.target) && e.target !== logoBtn) {
               controlPanel.style.display = 'none';
           }
       });
   }

   // Dark Mode 状态初始化与切换
   const savedTheme = localStorage.getItem('theme');
   if (savedTheme === 'light') {
       document.body.classList.add('light-mode');
       if (darkModeToggle) darkModeToggle.checked = false;
   }

   if (darkModeToggle) {
       darkModeToggle.addEventListener('change', (e) => {
           if (e.target.checked) {
               document.body.classList.remove('light-mode');
               localStorage.setItem('theme', 'dark');
           } else {
               document.body.classList.add('light-mode');
               localStorage.setItem('theme', 'light');
           }
       });
   }
});

       function showProject(regionKey) {
           currentRegion = regionKey;
           const data = projectData[regionKey];
           if (!data) return;

           const card = document.getElementById('project-detail');
           document.getElementById('project-title').innerText = data.title;
           document.getElementById('project-desc').innerText = data.desc;

           card.style.display = 'block';
           card.scrollIntoView({ behavior: 'smooth' });
       }

       function openOverviewModal() {
           if (!currentRegion) return;
           const data = projectData[currentRegion];
           
           document.getElementById('modalTitle').innerText = data.title + " - Overview";
           document.getElementById('modalDescription').innerText = data.desc;

           const gallery = document.getElementById('modalGallery');
           gallery.innerHTML = data.images.map(img => `<img src="${img}" alt="Project Photo">`).join('');

           document.getElementById('overviewModal').style.display = 'flex';
       }

       function closeOverviewModal(e) {
           document.getElementById('overviewModal').style.display = 'none';

// 全局火箭光标跟随逻辑
const rocketCursor = document.getElementById('rocketCursor');

// 仅在非触屏设备（有鼠标的设备）上启用火箭跟随
if (rocketCursor && window.matchMedia('(pointer: fine)').matches) {
   document.addEventListener('mousemove', (e) => {
       rocketCursor.style.left = `${e.clientX}px`;
       rocketCursor.style.top = `${e.clientY}px`;
   });
} else if (rocketCursor) {
   // 移动端/触屏设备直接隐藏
   rocketCursor.style.display = 'none';
}

二、 完整版 javascript.js
这个文件统领全站的全局互动：火箭光标（含触屏设备自动隐藏/桌面鼠标悬停变大）、Command Center 左上角控制面板、深色/浅色模式切换 以及 主页 2D 太阳系 Canvas 动画渲染。
直接用以下完整代码替换你的 javascript.js 文件：
/* ==========================================================================
  Jack's Portfolio - Global JavaScript Engine
  ========================================================================== */

document.addEventListener('DOMContentLoaded', () => {

   // ----------------------------------------------------------------------
   // 1. 全局火箭光标逻辑 (Rocket Cursor)
   // ----------------------------------------------------------------------
   const rocket = document.getElementById('rocketCursor');
   
   // 仅在配备鼠标的设备上开启光标跟随，移动端触屏设备直接隐藏
   if (rocket && window.matchMedia('(pointer: fine)').matches) {
       window.addEventListener('mousemove', (e) => {
           rocket.style.left = `${e.clientX}px`;
           rocket.style.top = `${e.clientY}px`;
       });

       // 悬停交互元素时变大提示
       const interactiveSelector = 'a, button, .map-marker, input, .switch, [role="button"]';
       document.addEventListener('mouseover', (e) => {
           if (e.target.closest(interactiveSelector)) rocket.classList.add('hovering');
       });
       document.addEventListener('mouseout', (e) => {
           if (e.target.closest(interactiveSelector)) rocket.classList.remove('hovering');
       });
   } else if (rocket) {
       rocket.style.display = 'none';
   }


   // ----------------------------------------------------------------------
   // 2. 左上角 Command Center 控制面板
   // ----------------------------------------------------------------------
   const logoBtn = document.getElementById('logoControlBtn');
   const controlPanel = document.getElementById('controlPanel');
   const darkModeToggle = document.getElementById('darkModeToggle');

   if (logoBtn && controlPanel) {
       logoBtn.addEventListener('click', (e) => {
           e.stopPropagation();
           const isVisible = controlPanel.style.display === 'block';
           controlPanel.style.display = isVisible ? 'none' : 'block';
       });

       // 点击外部区域自动关闭控制面板
       document.addEventListener('click', (e) => {
           if (!controlPanel.contains(e.target) && e.target !== logoBtn) {
               controlPanel.style.display = 'none';
           }
       });
   }

   // 主题本地化记忆 (Local Storage Theme State)
   const savedTheme = localStorage.getItem('theme');
   if (savedTheme === 'light') {
       document.body.classList.add('light-mode');
       if (darkModeToggle) darkModeToggle.checked = false;
   }

   if (darkModeToggle) {
       darkModeToggle.addEventListener('change', (e) => {
           if (e.target.checked) {
               document.body.classList.remove('light-mode');
               localStorage.setItem('theme', 'dark');
           } else {
               document.body.classList.add('light-mode');
               localStorage.setItem('theme', 'light');
           }
       });
   }


   // ----------------------------------------------------------------------
   // 3. 太阳系 Canvas 2D 互动渲染 (仅在存在 solarCanvas 的 index.html 生效)
   // ----------------------------------------------------------------------
   const canvas = document.getElementById('solarCanvas');
   if (canvas) {
       const ctx = canvas.getContext('2d');
       let width = canvas.width = window.innerWidth;
       let height = canvas.height = window.innerHeight;

       window.addEventListener('resize', () => {
           width = canvas.width = window.innerWidth;
           height = canvas.height = window.innerHeight;
       });

       // 太阳系星球配置
       const sun = { x: width / 2, y: height / 2, radius: 35, color: '#f59e0b' };
       
       const planets = [
           { name: 'Mercury', radius: 6, orbit: 80, speed: 0.02, angle: 0, color: '#94a3b8' },
           { name: 'Venus', radius: 10, orbit: 130, speed: 0.015, angle: 1, color: '#f39c12' },
           { name: 'Earth', radius: 12, orbit: 190, speed: 0.01, angle: 2, color: '#38bdf8' },
           { name: 'Mars', radius: 8, orbit: 250, speed: 0.008, angle: 3, color: '#ef4444' },
           { name: 'Jupiter', radius: 20, orbit: 330, speed: 0.005, angle: 4, color: '#eab308' }
       ];

       // 渲染循环
       function animate() {
           ctx.clearRect(0, 0, width, height);

           const centerX = width / 2;
           const centerY = height / 2;

           // 1. 绘制太阳
           ctx.beginPath();
           ctx.arc(centerX, centerY, sun.radius, 0, Math.PI * 2);
           ctx.fillStyle = sun.color;
           ctx.shadowBlur = 30;
           ctx.shadowColor = sun.color;
           ctx.fill();
           ctx.shadowBlur = 0; // 重置阴影

           // 2. 绘制轨道与行星
           planets.forEach(p => {
               p.angle += p.speed;

               // 绘制轨迹线
               ctx.beginPath();
               ctx.arc(centerX, centerY, p.orbit, 0, Math.PI * 2);
               ctx.strokeStyle = 'rgba(255, 255, 255, 0.08)';
               ctx.lineWidth = 1;
               ctx.stroke();

               // 计算行星坐标
               const px = centerX + Math.cos(p.angle) * p.orbit;
               const py = centerY + Math.sin(p.angle) * p.orbit;

               // 绘制行星本体
               ctx.beginPath();
               ctx.arc(px, py, p.radius, 0, Math.PI * 2);
               ctx.fillStyle = p.color;
               ctx.fill();
           });

           requestAnimationFrame(animate);
       }

       animate();
   }
});
